PROJECT OVERVIEW
------------------------------------------------------------
Graa AI turns any learning goal into a day-by-day roadmap, then for each day
serves curated content (a relevant YouTube video + docs + an AI-written lesson),
a multiple-choice quiz, and a hands-on practice task. A learner can only unlock
the next day after completing the current day's task. An AI mentor ("Graa") is
available on every page and answers grounded in the learner's own curriculum.

Neon PostgreSQL with the pgvector extension, NextAuth (credentials + Google OAuth).
AI models: Groq (Llama 3.1 8B / 3.3 70B) for all text generation, with NVIDIA NIM
as a cross-provider fallback; Jina embeddings (v3) for RAG. Code execution via the
Paiza API; content sourced free from DuckDuckGo + YouTube. Everything runs on free
tiers ($0 cost).

function modelChain(primary: string): string[] {
  return [primary, 'llama-3.1-8b-instant', 'llama-3.3-70b-versatile']
    .filter((m, i, a) => a.indexOf(m) === i)
}

async function tryGroq(messages, options): Promise<string | null> {
  const keys = groqKeys()                       // GROQ_API_KEY, GROQ_API_KEY2, …
  for (let round = 0; round < 2; round++) {
    for (const model of modelChain(options.model || DEFAULT_MODEL)) {
      for (const key of keys) {
        try {
          const r = await axios.post(GROQ_URL, { model, messages, ... },
            { headers: { Authorization: `Bearer ${key}` } })
          return r.data.choices?.[0]?.message?.content?.trim() || ''
        } catch (err) {
          const status = err.response?.status
          if (status === 429 || status >= 500) continue   // try next key / model
          return null
        }
      }
    }
    await sleep(1200 * (round + 1))
  }
  return null                                   // all Groq attempts rate-limited
}

async function createChatCompletion(messages, options): Promise<string> {
  const groq = await tryGroq(messages, options)
  if (groq !== null) return groq
  const nvidia = await tryNvidia(messages, options)   // NVIDIA NIM, separate quota
  if (nvidia !== null) return nvidia
  throw new Error('All free-tier AI providers are rate-limited. Try again later.')
}

FEATURE 4 — RAG: pgvector + local-cost-free hosted embeddings
Each day's content is embedded (Jina, 384-dim) into a pgvector column; the mentor

export async function retrieve(goalId: string, query: string, k = 5) {
  if (!embeddingsEnabled) return []
  await ensureStore()
  const vec = toVectorLiteral(await embed(query))
  return prisma.$queryRawUnsafe(
    `SELECT content, source, 1 - (embedding <=> $1::vector) AS score
       FROM content_chunks
      WHERE goal_id = $2
      ORDER BY embedding <=> $1::vector
      LIMIT $3`,
    vec, goalId, k)
}
